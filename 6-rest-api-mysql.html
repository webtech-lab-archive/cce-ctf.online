Program 6 — REST API + MySQL This program creates a REST API using Node.js + Express.js and connects it to MySQL for CRUD operations: GET, POST, PUT, DELETE. 1. Install packages npm init -y npm install express mysql2 cors 2. MySQL CREATE DATABASE studentdb; USE studentdb; CREATE TABLE students( id INT AUTO_INCREMENT PRIMARY KEY, name VARCHAR(100), email VARCHAR(100), department VARCHAR(100) ); 3. server.js — Short code const express = require("express"); const mysql = require("mysql2"); const cors = require("cors"); const app = express(); app.use(express.json()); app.use(cors()); const db = mysql.createConnection({ host:"localhost", user:"root", password:"", database:"studentdb" }); db.connect(err => { if(err) console.log("DB connection failed"); else console.log("Connected to MySQL"); }); // GET app.get("/students",(req,res)=>{ db.query("SELECT * FROM students",(err,result)=>{ if(err) res.status(500).send(err); else res.json(result); }); }); // POST app.post("/students",(req,res)=>{ const {name,email,department}=req.body; db.query( "INSERT INTO students(name,email,department) VALUES(?,?,?)", [name,email,department], (err,result)=>{ if(err) res.status(500).send(err); else res.json({message:"Student added",id:result.insertId}); } ); }); // PUT app.put("/students/:id",(req,res)=>{ const {name,email,department}=req.body; db.query( "UPDATE students SET name=?,email=?,department=? WHERE id=?", [name,email,department,req.params.id], (err)=>{ if(err) res.status(500).send(err); else res.json({message:"Student updated"}); } ); }); // DELETE app.delete("/students/:id",(req,res)=>{ db.query( "DELETE FROM students WHERE id=?", [req.params.id], (err)=>{ if(err) res.status(500).send(err); else res.json({message:"Student deleted"}); } ); }); app.listen(3000,()=>{ console.log("Server running on port 3000"); }); ▶️ Run node server.js
